# Areas (Room Grouping) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add polygon area overlays to floorplans that group item placements for per-room invoicing and a pivot-table DOCX export page.

**Architecture:** Areas are a new placement type (`type = 'area'`) stored in the existing `placements` table with nullable `bom_id`. Area-specific data (name, color, opacity, polygon vertices) lives in separate `area_properties` and `area_vertices` tables. Item placements reference their containing area via `area_id`. A new Areas tab in the right panel provides drag-to-create and area management. Canvas renders areas as SVG polygon overlays below item placements.

**Tech Stack:** Deno + Hono + SQLite (backend), React 18 + Vite + TypeScript + Tailwind + shadcn/ui + dnd-kit + SVG (frontend), docx library (export)

**Spec:** `docs/superpowers/specs/2026-03-24-areas-design.md`

```typescript
import { getDb } from '../config/database.ts';
import type { Area, CreateAreaDTO, UpdateAreaDTO, AreaVertex } from '../models/index.ts';

export class AreaRepository {
  findByFloorplan(floorplanId: number): Promise<Area[]> {
    const areas = getDb().queryEntries<Record<string, unknown>>(`
      SELECT p.id, p.floorplan_id, p.x, p.y, p.width, p.height, p.created_at,
             ap.name, ap.color, ap.opacity, ap.updated_at,
             (SELECT COUNT(*) FROM placements ip WHERE ip.area_id = p.id AND ip.type = 'item') as device_count
      FROM placements p
      JOIN area_properties ap ON ap.placement_id = p.id
      WHERE p.floorplan_id = ? AND p.type = 'area'
      ORDER BY ap.name ASC
    `, [floorplanId]);

    const result: Area[] = areas.map(a => {
      const vertices = getDb().queryEntries<AreaVertex>(`
        SELECT id, placement_id, vertex_index, x, y
        FROM area_vertices
        WHERE placement_id = ?
        ORDER BY vertex_index ASC
      `, [a.id]);

      return {
        id: a.id as number,
        floorplan_id: a.floorplan_id as number,
        x: a.x as number,
        y: a.y as number,
        width: a.width as number,
        height: a.height as number,
        name: a.name as string,
        color: a.color as string,
        opacity: a.opacity as number,
        vertices: vertices as AreaVertex[],
        device_count: a.device_count as number,
        created_at: a.created_at as string,
        updated_at: a.updated_at as string,
      };
    });

    return Promise.resolve(result);
  }

  findById(id: number): Promise<Area | null> {
    const areas = getDb().queryEntries<Record<string, unknown>>(`
      SELECT p.id, p.floorplan_id, p.x, p.y, p.width, p.height, p.created_at,
             ap.name, ap.color, ap.opacity, ap.updated_at,
             (SELECT COUNT(*) FROM placements ip WHERE ip.area_id = p.id AND ip.type = 'item') as device_count
      FROM placements p
      JOIN area_properties ap ON ap.placement_id = p.id
      WHERE p.id = ? AND p.type = 'area'
    `, [id]);

    if (areas.length === 0) return Promise.resolve(null);

    const a = areas[0];
    const vertices = getDb().queryEntries<AreaVertex>(`
      SELECT id, placement_id, vertex_index, x, y
      FROM area_vertices
      WHERE placement_id = ?
      ORDER BY vertex_index ASC
    `, [id]);

    return Promise.resolve({
      id: a.id as number,
      floorplan_id: a.floorplan_id as number,
      x: a.x as number,
      y: a.y as number,
      width: a.width as number,
      height: a.height as number,
      name: a.name as string,
      color: a.color as string,
      opacity: a.opacity as number,
      vertices: vertices as AreaVertex[],
      device_count: a.device_count as number,
      created_at: a.created_at as string,
      updated_at: a.updated_at as string,
    });
  }

  create(data: CreateAreaDTO): Promise<Area> {
    const db = getDb();

    db.query("BEGIN");
    try {
      // Create placement
      db.query(`
        INSERT INTO placements (bom_id, floorplan_id, type, x, y, width, height, rotation)
        VALUES (NULL, ?, 'area', ?, ?, ?, ?, 0)
      `, [data.floorplan_id, data.x, data.y, data.width, data.height]);

      const placementId = db.lastInsertRowId;

      // Create area properties
      db.query(`
        INSERT INTO area_properties (placement_id, name, color, opacity)
        VALUES (?, ?, ?, ?)
      `, [
        placementId,
        data.name ?? 'New Area',
        data.color ?? '#3b82f6',
        data.opacity ?? 0.1,
      ]);

      // Create default rectangle vertices (4 corners)
      const vertices = [
        { x: data.x, y: data.y },
        { x: data.x + data.width, y: data.y },
        { x: data.x + data.width, y: data.y + data.height },
        { x: data.x, y: data.y + data.height },
      ];

      vertices.forEach((v, i) => {
        db.query(`
          INSERT INTO area_vertices (placement_id, vertex_index, x, y)
          VALUES (?, ?, ?, ?)
        `, [placementId, i, v.x, v.y]);
      });

      db.query("COMMIT");
      return this.findById(Number(placementId)) as Promise<Area>;
    } catch (error) {
      db.query("ROLLBACK");
      throw error;
    }
  }

  updateProperties(id: number, data: UpdateAreaDTO): Promise<Area | null> {
    const sets: string[] = [];
    const values: (string | number)[] = [];

    if (data.name !== undefined) { sets.push('name = ?'); values.push(data.name); }
    if (data.color !== undefined) { sets.push('color = ?'); values.push(data.color); }
    if (data.opacity !== undefined) { sets.push('opacity = ?'); values.push(data.opacity); }

    if (sets.length === 0) return this.findById(id);

    sets.push("updated_at = CURRENT_TIMESTAMP");
    values.push(id);

    getDb().query(`
      UPDATE area_properties SET ${sets.join(', ')} WHERE placement_id = ?
    `, values);

    return this.findById(id);
  }

  updateVertices(id: number, vertices: { x: number; y: number }[]): Promise<Area | null> {
    const db = getDb();

    db.query("BEGIN");
    try {
      // Delete existing vertices
      db.query(`DELETE FROM area_vertices WHERE placement_id = ?`, [id]);

      // Insert new vertices
      vertices.forEach((v, i) => {
        db.query(`
          INSERT INTO area_vertices (placement_id, vertex_index, x, y)
          VALUES (?, ?, ?, ?)
        `, [id, i, v.x, v.y]);
      });

      // Recompute bounding box
      const minX = Math.min(...vertices.map(v => v.x));
      const minY = Math.min(...vertices.map(v => v.y));
      const maxX = Math.max(...vertices.map(v => v.x));
      const maxY = Math.max(...vertices.map(v => v.y));

      db.query(`
        UPDATE placements SET x = ?, y = ?, width = ?, height = ?
        WHERE id = ? AND type = 'area'
      `, [minX, minY, maxX - minX, maxY - minY, id]);

      db.query("COMMIT");
    } catch (error) {
      db.query("ROLLBACK");
      throw error;
    }

    return this.findById(id);
  }

  delete(id: number): Promise<void> {
    const db = getDb();
    // Nullify area_id on contained item placements and their BOM entries
    db.query(`UPDATE placements SET area_id = NULL WHERE area_id = ?`, [id]);
    db.query(`UPDATE project_bom SET area_id = NULL WHERE area_id = ?`, [id]);
    // Delete area placement (cascades to area_properties and area_vertices)
    db.query(`DELETE FROM placements WHERE id = ? AND type = 'area'`, [id]);
    return Promise.resolve();
  }

  /** Assign an item placement to an area based on point-in-polygon */
  assignPlacementToArea(placementId: number, areaId: number | null): Promise<void> {
    const db = getDb();
    db.query(`UPDATE placements SET area_id = ? WHERE id = ?`, [areaId, placementId]);
    // Sync to BOM
    const placement = db.queryEntries<{ bom_id: number | null }>(`
      SELECT bom_id FROM placements WHERE id = ?
    `, [placementId]);
    if (placement.length > 0 && placement[0].bom_id) {
      db.query(`UPDATE project_bom SET area_id = ? WHERE id = ?`, [areaId, placement[0].bom_id]);
    }
    return Promise.resolve();
  }

  /** Get all areas for a floorplan (lightweight, for containment checks) */
  findVerticesForFloorplan(floorplanId: number): Promise<{ areaId: number; vertices: { x: number; y: number }[] }[]> {
    const areas = getDb().queryEntries<{ id: number }>(`
      SELECT id FROM placements WHERE floorplan_id = ? AND type = 'area'
    `, [floorplanId]);

    const result = areas.map(a => {
      const vertices = getDb().queryEntries<{ x: number; y: number }>(`
        SELECT x, y FROM area_vertices WHERE placement_id = ? ORDER BY vertex_index ASC
      `, [a.id]);
      return { areaId: a.id, vertices };
    });

    return Promise.resolve(result);
  }
}

export const areaRepository = new AreaRepository();
```

```typescript
import { Hono } from 'hono';
import { zValidator } from '@hono/zod-validator';
import { z } from 'zod';
import { areaRepository } from '../repositories/area.ts';
import { authMiddleware } from '../middleware/auth.ts';

const areaRoutes = new Hono();

const createAreaSchema = z.object({
  floorplan_id: z.number(),
  x: z.number(),
  y: z.number(),
  width: z.number().positive(),
  height: z.number().positive(),
  name: z.string().optional(),
  color: z.string().optional(),
  opacity: z.number().min(0).max(1).optional(),
});

const updateAreaSchema = z.object({
  name: z.string().optional(),
  color: z.string().optional(),
  opacity: z.number().min(0).max(1).optional(),
});

const updateVerticesSchema = z.object({
  vertices: z.array(z.object({
    x: z.number(),
    y: z.number(),
  })).min(3),
});

// GET /areas?floorplan_id=X
areaRoutes.get('/', authMiddleware, async (c) => {
  try {
    const floorplanId = c.req.query('floorplan_id');
    if (!floorplanId) {
      return c.json({ error: 'floorplan_id is required' }, 400);
    }
    const areas = await areaRepository.findByFloorplan(parseInt(floorplanId));
    return c.json({ data: areas });
  } catch (error) {
    console.error('Error fetching areas:', error);
    return c.json({ error: 'Failed to fetch areas' }, 500);
  }
});

// GET /areas/:id
areaRoutes.get('/:id', authMiddleware, async (c) => {
  try {
    const id = parseInt(c.req.param('id'));
    const area = await areaRepository.findById(id);
    if (!area) return c.json({ error: 'Area not found' }, 404);
    return c.json({ data: area });
  } catch (error) {
    console.error('Error fetching area:', error);
    return c.json({ error: 'Failed to fetch area' }, 500);
  }
});

// POST /areas
areaRoutes.post('/', authMiddleware, zValidator('json', createAreaSchema), async (c) => {
  try {
    const data = c.req.valid('json');
    const area = await areaRepository.create(data);
    return c.json({ data: area, message: 'Area created' }, 201);
  } catch (error) {
    console.error('Error creating area:', error);
    return c.json({ error: 'Failed to create area' }, 500);
  }
});

// PUT /areas/:id
areaRoutes.put('/:id', authMiddleware, zValidator('json', updateAreaSchema), async (c) => {
  try {
    const id = parseInt(c.req.param('id'));
    const data = c.req.valid('json');
    const area = await areaRepository.updateProperties(id, data);
    if (!area) return c.json({ error: 'Area not found' }, 404);
    return c.json({ data: area });
  } catch (error) {
    console.error('Error updating area:', error);
    return c.json({ error: 'Failed to update area' }, 500);
  }
});

// PUT /areas/:id/vertices
areaRoutes.put('/:id/vertices', authMiddleware, zValidator('json', updateVerticesSchema), async (c) => {
  try {
    const id = parseInt(c.req.param('id'));
    const { vertices } = c.req.valid('json');
    const area = await areaRepository.updateVertices(id, vertices);
    if (!area) return c.json({ error: 'Area not found' }, 404);
    return c.json({ data: area });
  } catch (error) {
    console.error('Error updating vertices:', error);
    return c.json({ error: 'Failed to update vertices' }, 500);
  }
});

// DELETE /areas/:id
areaRoutes.delete('/:id', authMiddleware, async (c) => {
  try {
    const id = parseInt(c.req.param('id'));
    const area = await areaRepository.findById(id);
    if (!area) return c.json({ error: 'Area not found' }, 404);
    await areaRepository.delete(id);
    return c.json({ message: 'Area deleted' });
  } catch (error) {
    console.error('Error deleting area:', error);
    return c.json({ error: 'Failed to delete area' }, 500);
  }
});

export default areaRoutes;
```

```typescript
import { useState, useCallback, useRef, useEffect } from 'react';
import { areaService } from '@/services/area';
import type { Area, CreateAreaDTO, UpdateAreaDTO } from '@/services/area';

interface UseAreasProps {
  activeFloorplanId: number | null;
}

interface UseAreasReturn {
  areas: Area[];
  isLoading: boolean;
  fetchAreas: (floorplanId: number) => Promise<void>;
  createArea: (data: CreateAreaDTO) => Promise<Area | null>;
  updateArea: (id: number, data: UpdateAreaDTO) => Promise<Area | null>;
  updateVertices: (id: number, vertices: { x: number; y: number }[]) => Promise<Area | null>;
  deleteArea: (id: number) => Promise<void>;
  selectedAreaId: number | null;
  setSelectedAreaId: (id: number | null) => void;
}

export function useAreas({ activeFloorplanId }: UseAreasProps): UseAreasReturn {
  const [areas, setAreas] = useState<Area[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [selectedAreaId, setSelectedAreaId] = useState<number | null>(null);
  const abortRef = useRef<AbortController | null>(null);

  const fetchAreas = useCallback(async (floorplanId: number) => {
    abortRef.current?.abort();
    const controller = new AbortController();
    abortRef.current = controller;

    setIsLoading(true);
    try {
      const data = await areaService.getByFloorplan(floorplanId, controller.signal);
      setAreas(data);
    } catch (error) {
      if (error instanceof DOMException && error.name === 'AbortError') return;
      console.error('Error fetching areas:', error);
    } finally {
      setIsLoading(false);
    }
  }, []);

  const createArea = useCallback(async (data: CreateAreaDTO): Promise<Area | null> => {
    try {
      const area = await areaService.create(data);
      setAreas(prev => [...prev, area]);
      return area;
    } catch (error) {
      console.error('Error creating area:', error);
      return null;
    }
  }, []);

  const updateArea = useCallback(async (id: number, data: UpdateAreaDTO): Promise<Area | null> => {
    try {
      const area = await areaService.update(id, data);
      setAreas(prev => prev.map(a => a.id === id ? area : a));
      return area;
    } catch (error) {
      console.error('Error updating area:', error);
      return null;
    }
  }, []);

  const updateVertices = useCallback(async (id: number, vertices: { x: number; y: number }[]): Promise<Area | null> => {
    try {
      const area = await areaService.updateVertices(id, vertices);
      setAreas(prev => prev.map(a => a.id === id ? area : a));
      return area;
    } catch (error) {
      console.error('Error updating vertices:', error);
      return null;
    }
  }, []);

  const deleteArea = useCallback(async (id: number) => {
    try {
      await areaService.delete(id);
      setAreas(prev => prev.filter(a => a.id !== id));
      if (selectedAreaId === id) setSelectedAreaId(null);
    } catch (error) {
      console.error('Error deleting area:', error);
    }
  }, [selectedAreaId]);

  useEffect(() => {
    if (activeFloorplanId) {
      fetchAreas(activeFloorplanId);
    } else {
      setAreas([]);
    }
  }, [activeFloorplanId, fetchAreas]);

  return {
    areas, isLoading, fetchAreas,
    createArea, updateArea, updateVertices, deleteArea,
    selectedAreaId, setSelectedAreaId,
  };
}
```

```typescript
}: AreaPolygonProps) {
  const [isDragging, setIsDragging] = useState(false);
  const dragStartRef = useRef<{ x: number; y: number } | null>(null);

  const points = area.vertices.map(v => `${v.x},${v.y}`).join(' ');

  // Compute label position (first vertex + offset)
  const labelX = area.vertices.length > 0 ? Math.min(...area.vertices.map(v => v.x)) + 8 : 0;
  const labelY = area.vertices.length > 0 ? Math.min(...area.vertices.map(v => v.y)) + 18 : 0;

  const handlePolygonMouseDown = useCallback((e: React.MouseEvent) => {
    if (e.button !== 0) return;
    e.stopPropagation();
    onSelect(area.id);

    dragStartRef.current = { x: e.clientX, y: e.clientY };
    setIsDragging(true);

    const handleMouseMove = (moveEvent: MouseEvent) => {
      if (!dragStartRef.current) return;
      const dx = (moveEvent.clientX - dragStartRef.current.x) / scale;
      const dy = (moveEvent.clientY - dragStartRef.current.y) / scale;
      dragStartRef.current = { x: moveEvent.clientX, y: moveEvent.clientY };
      onMove(area.id, dx, dy);
    };

    const handleMouseUp = () => {
      setIsDragging(false);
      dragStartRef.current = null;
      onVerticesCommit(area.id);
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('mouseup', handleMouseUp);
    };

    window.addEventListener('mousemove', handleMouseMove);
    window.addEventListener('mouseup', handleMouseUp);
  }, [area.id, scale, onSelect, onMove, onVerticesCommit]);

  const handleVertexMouseDown = useCallback((e: React.MouseEvent, vertexIndex: number) => {
    e.stopPropagation();
    e.preventDefault();

    const handleMouseMove = (moveEvent: MouseEvent) => {
      // Convert screen coords to canvas coords using the SVG's CTM
      const svg = (e.target as SVGElement).ownerSVGElement;
      if (!svg) return;
      const pt = svg.createSVGPoint();
      pt.x = moveEvent.clientX;
      pt.y = moveEvent.clientY;
      const svgP = pt.matrixTransform(svg.getScreenCTM()!.inverse());
      onVertexMove(area.id, vertexIndex, svgP.x, svgP.y);
    };

    const handleMouseUp = () => {
      onVerticesCommit(area.id);
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('mouseup', handleMouseUp);
    };

    window.addEventListener('mousemove', handleMouseMove);
    window.addEventListener('mouseup', handleMouseUp);
  }, [area.id, onVertexMove, onVerticesCommit]);

  const handleEdgeClick = useCallback((e: React.MouseEvent, edgeIndex: number) => {
    if (!e.ctrlKey && !e.metaKey) return;
    e.stopPropagation();

    const svg = (e.target as SVGElement).ownerSVGElement;
    if (!svg) return;
    const pt = svg.createSVGPoint();
    pt.x = e.clientX;
    pt.y = e.clientY;
    const svgP = pt.matrixTransform(svg.getScreenCTM()!.inverse());
    onVertexAdd(area.id, edgeIndex, svgP.x, svgP.y);
  }, [area.id, onVertexAdd]);

  const handleSize = 8 / scale;

  return (
    <g>
      {/* Fill polygon */}
      <polygon
        points={points}
        fill={area.color}
        fillOpacity={area.opacity}
        stroke={area.color}
        strokeWidth={2 / scale}
        strokeOpacity={0.8}
        style={{ cursor: isDragging ? 'grabbing' : 'grab' }}
        onMouseDown={handlePolygonMouseDown}
      />

      {/* Name label */}
      <text
        x={labelX}
        y={labelY}
        fill={area.color}
        fontSize={14 / scale}
        fontWeight="bold"
        opacity={0.9}
        style={{ pointerEvents: 'none', userSelect: 'none' }}
      >
        {area.name}
      </text>

      {/* Edge click targets for Ctrl+click vertex insertion */}
      {isSelected && area.vertices.map((v, i) => {
        const next = area.vertices[(i + 1) % area.vertices.length];
        return (
          <line
            key={`edge-${i}`}
            x1={v.x} y1={v.y} x2={next.x} y2={next.y}
            stroke="transparent"
            strokeWidth={12 / scale}
            style={{ cursor: 'crosshair' }}
            onClick={(e) => handleEdgeClick(e, i)}
          />
        );
      })}

      {/* Vertex handles */}
      {isSelected && area.vertices.map((v, i) => (
        <rect
          key={`vertex-${i}`}
          x={v.x - handleSize / 2}
          y={v.y - handleSize / 2}
          width={handleSize}
          height={handleSize}
          fill="white"
          stroke={area.color}
          strokeWidth={1.5 / scale}
          style={{ cursor: 'move' }}
          onMouseDown={(e) => handleVertexMouseDown(e, i)}
        />
      ))}

      {/* Action buttons when selected */}
      {isSelected && (
        <g>
          <foreignObject x={labelX} y={labelY + 8} width={60 / scale} height={24 / scale}>
            <div style={{ display: 'flex', gap: `${2 / scale}px`, transform: `scale(${1 / scale})`, transformOrigin: 'top left' }}>
              <button
                onClick={(e) => { e.stopPropagation(); onEdit(area.id); }}
                className="px-1.5 py-0.5 text-xs bg-card border rounded hover:bg-accent"
                title="Edit"
              >
                Edit
              </button>
              <button
                onClick={(e) => { e.stopPropagation(); onDelete(area.id); }}
                className="px-1.5 py-0.5 text-xs bg-card border rounded hover:bg-destructive hover:text-destructive-foreground"
                title="Delete"
              >
                Del
              </button>
            </div>
          </foreignObject>
        </g>
      )}
    </g>
  );
}
```

```typescript
/** Ray casting algorithm for point-in-polygon */
export function pointInPolygon(px: number, py: number, vertices: { x: number; y: number }[]): boolean {
  let inside = false;
  for (let i = 0, j = vertices.length - 1; i < vertices.length; j = i++) {
    const xi = vertices[i].x, yi = vertices[i].y;
    const xj = vertices[j].x, yj = vertices[j].y;

    const intersect = ((yi > py) !== (yj > py))
      && (px < (xj - xi) * (py - yi) / (yj - yi) + xi);
    if (intersect) inside = !inside;
  }
  return inside;
}

/** Shoelace formula for polygon area */
export function polygonArea(vertices: { x: number; y: number }[]): number {
  let area = 0;
  for (let i = 0, j = vertices.length - 1; i < vertices.length; j = i++) {
    area += (vertices[j].x + vertices[i].x) * (vertices[j].y - vertices[i].y);
  }
  return Math.abs(area / 2);
}

/** Find the containing area for a placement center point. Smallest area wins. */
export function findContainingArea(
  px: number, py: number,
  areas: { id: number; vertices: { x: number; y: number }[] }[]
): number | null {
  let bestId: number | null = null;
  let bestArea = Infinity;

  for (const area of areas) {
    if (pointInPolygon(px, py, area.vertices)) {
      const a = polygonArea(area.vertices);
      if (a < bestArea) {
        bestArea = a;
        bestId = area.id;
      }
    }
  }

  return bestId;
}
```
